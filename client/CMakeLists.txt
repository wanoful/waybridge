  foreach(target waybridge-client waybridge-readonly-probe waybridge-decode-probe waybridge-image-probe waybridge-resize-probe waybridge-hdr-alpha-probe waybridge-udp-probe waybridge-gamepad-probe waybridge-drop-lifetime-probe)
