cmake_minimum_required(VERSION 3.20)
project(waybridge_client VERSION 0.3.0 LANGUAGES CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_executable(waybridge-client main.cpp)
target_compile_features(waybridge-client PRIVATE cxx_std_17)
target_compile_definitions(waybridge-client PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_definitions(waybridge-client PRIVATE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_compile_definitions(waybridge-client PRIVATE WAYBRIDGE_VERSION="${PROJECT_VERSION}")
target_link_libraries(waybridge-client PRIVATE ws2_32 gdiplus ole32 crypt32 shell32 uuid d3d11 dxgi dcomp d3dcompiler mfuuid wmcodecdspuuid strmiids oleaut32 windowscodecs)
add_executable(waybridge-decode-probe EXCLUDE_FROM_ALL ../tests/windows_decode_probe.cpp)
target_compile_features(waybridge-decode-probe PRIVATE cxx_std_17)
target_compile_definitions(waybridge-decode-probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_include_directories(waybridge-decode-probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(waybridge-decode-probe PRIVATE gdiplus ole32 uuid d3d11 dxgi dcomp mfuuid wmcodecdspuuid strmiids oleaut32 windowscodecs)
add_executable(waybridge-image-probe EXCLUDE_FROM_ALL ../tests/windows_image_probe.cpp)
target_compile_features(waybridge-image-probe PRIVATE cxx_std_17)
target_compile_definitions(waybridge-image-probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(waybridge-image-probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(waybridge-image-probe PRIVATE gdiplus ole32 windowscodecs uuid)
add_executable(waybridge-resize-probe EXCLUDE_FROM_ALL ../tests/windows_resize_probe.cpp)
target_compile_features(waybridge-resize-probe PRIVATE cxx_std_17)
target_compile_definitions(waybridge-resize-probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_include_directories(waybridge-resize-probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(waybridge-resize-probe PRIVATE ole32 uuid d3d11 dxgi dcomp d3dcompiler mfuuid wmcodecdspuuid strmiids oleaut32)
if(MINGW)
  foreach(target waybridge-client waybridge-decode-probe waybridge-image-probe waybridge-resize-probe)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(${target} PRIVATE -static)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
      target_link_options(${target} PRIVATE -static-libgcc -static-libstdc++)
    endif()
  endforeach()
  target_link_options(waybridge-client PRIVATE -municode)
endif()
